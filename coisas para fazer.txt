para fazer as coisas abaixo, olhar o tutorial que o mano claudio me deu
sudo apt install openssh-server
sudo systemctl enable ssh --now

sudo mkdir -p /var/www/aula-marcao
sudo chown -r MEU-USUARIO:MEU_USUARIO /var/www/aula-marcao

sudo nano /etc/apache2/sites-available/aula-marcao.conf
<VirtualHost *:8080>
    DocumentRoot /var/www/aula-marcao
    <Directory /var/www/aula-marcao>
        AllowOverride All
        Require all granted
    </Directory>
</VirtualHost>


sudo sed -i '/Listen 80/a Listen 8080' /etc/apache2/portas.conf
sudo a2ensite aula-marcao.conf
sudo systemctl reload apache2

sudo usermod -aG MEU_USUARIO www-data
sudo chmod -R 750 /var/www/aula-marcao
